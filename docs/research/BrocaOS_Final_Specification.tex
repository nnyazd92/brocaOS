\begin{abstract}
This document specifies BrocaOS, a cognitive operating system for agentic AI, with precise formal models, explicit cryptographic assumptions, and bounded verification. We present: (1) an action-instance safety model with history-based precedence tracking, (2) cryptographic token binding to canonical action digests, (3) bounded model checking (B=100) with history variable encoding, (4) sound abstraction via simulation relations with runtime monitors, (5) reproducible experimental protocols with proper statistical methods, and (6) explicit threat models and limitations. All claims are rigorously scoped with clear boundaries between verified properties (over finite abstractions), empirical measurements (under controlled conditions), and design specifications.
\end{abstract}

\section{Action-Instance Safety with History Tracking}

\subsection{Core Definitions}

\begin{definition}[Action Instance]
An action instance $a = (\text{id}, \text{type}, \text{args}, \text{context\_hash}, \text{nonce})$ where:
\begin{itemize}
    \item $\text{id} \in \mathbb{N}$: unique monotonic identifier
    \item $\text{type} \in \mathcal{T}$: action type (tool call, query, etc.)
    \item $\text{args} \in \mathcal{A}$: serialized parameter values
    \item $\text{context\_hash} = H(\text{canonicalize}(\text{env\_snapshot}))$: digest of relevant context
    \item $\text{nonce} \in \{0,1\}^{128}$: freshness randomness
\end{itemize}
\end{definition}

\begin{definition}[Action Digest]
\[
\mathrm{dig}(a) = \mathrm{HMAC}_{K}(\text{id} \| \text{type} \| \text{args} \| \text{context\_hash} \| \text{nonce})
\]
where $K$ is a system secret key and $\|$ denotes deterministic concatenation.
\end{definition}

\begin{definition}[Token with Cryptographic Binding]
A token $t = (\mathrm{dig}(a), \text{expiry}, \text{signature})$ where:
\begin{itemize}
    \item $\mathrm{dig}(a)$: action digest from Definition 2.2
    \item $\text{expiry} \in \mathbb{N}$: logical time validity bound
    \item $\text{signature} = \mathrm{HMAC}_{K'}(\mathrm{dig}(a) \| \text{expiry})$: cryptographic binding
\end{itemize}
\end{definition}

\subsection{Safety Predicates with History}

\begin{definition}[History Variable]
$H \subseteq \{\mathrm{dig}(a) : a \in \text{Actions}\}$ tracks digests of ever-approved actions.
\end{definition}

\begin{definition}[Safety Classification]
For action instance $a$:
\begin{itemize}
    \item $\text{irreversible}(a)$: action cannot be undone (predefined tool list)
    \item $\text{risk}(a) \in [0,1]$: estimated harm probability (heuristic + rules)
    \item $\text{unsafe}(a) \equiv \text{irreversible}(a) \land \text{risk}(a) > \tau$ where $\tau=0.7$
\end{itemize}
\end{definition}

\subsection{Formal State Machine with History}

\begin{spec}[Gating State Machine with History]
States $S = (\text{mode}, \text{pending}, \text{current}, H, k)$ where:
\begin{itemize}
    \item $\text{mode} \in \{\text{idle}, \text{pending}, \text{approved}, \text{executing}, \text{blocked}\}$
    \item $\text{pending} \subseteq \{\mathrm{dig}(a)\}$: pending action digests
    \item $\text{current} \in \{\mathrm{dig}(a)\} \cup \{\bot\}$: currently executing action
    \item $H \subseteq \{\mathrm{dig}(a)\}$: history of approved digests
    \item $k \in \mathbb{N}$: logical time step
\end{itemize}

Transitions:
\begin{align*}
& (\text{idle}, \_, \bot, H, k) \xrightarrow{\text{generate}(a)} (\text{pending}, \{\mathrm{dig}(a)\}, \bot, H, k+1) \\
& \quad \text{if } \text{unsafe}(a) \\
& (\text{pending}, P, \bot, H, k) \xrightarrow{\text{approve}(t)} (\text{approved}, P\setminus\{\mathrm{dig}(a)\}, \bot, H\cup\{\mathrm{dig}(a)\}, k+1) \\
& \quad \text{if } t.\mathrm{dig}(a) \in P \land \text{valid}(t,k) \\
& (\text{approved}, P, \bot, H, k) \xrightarrow{\text{execute}(a)} (\text{executing}, P, \mathrm{dig}(a), H, k+1) \\
& \quad \text{if } \mathrm{dig}(a) \in H \\
& (\text{idle}, \_, \bot, H, k) \xrightarrow{\text{generate}(a)} (\text{executing}, \emptyset, \mathrm{dig}(a), H, k+1) \\
& \quad \text{if } \neg\text{unsafe}(a) \\
& (\text{executing}, P, d, H, k) \xrightarrow{\text{done}} (\text{idle}, P, \bot, H, k+1) \\
& \text{any state} \xrightarrow{\text{timeout}} (\text{blocked}, \emptyset, \bot, H, k+1) \quad \text{if } k > \text{threshold}
\end{align*}
\end{spec}

\section{Formal Verification with History-Based Invariant}

\subsection{Verification Model}

\begin{definition}[Finite Bounded Kripke Structure]
$K_B = (S_B, S_0, R_B, L)$ where:
\begin{itemize}
    \item $S_B$: states from Spec. 2.3 with bounds: max 5 concurrent actions, $k \leq 100$
    \item $S_0 = \{(\text{idle}, \emptyset, \bot, \emptyset, 0)\}$
    \item $R_B \subseteq S_B \times S_B$: transitions from Spec. 2.3 respecting bounds
    \item $L: S_B \to 2^{AP}$ with $AP = \{\text{executing}(d), \text{unsafe}(d), d \in H\}$
\end{itemize}
\end{definition}

\subsection{History-Based Safety Invariant}

\begin{invariant}[Action-Instance Safety with History]
For all traces $\pi = s_0 s_1 \dots$ of $K_B$:
\[
\Box\left(\bigvee_{d \in \text{Digests}} \text{executing}(d) \land \text{unsafe}(d) \Rightarrow d \in H\right)
\]
where $H$ is the history component of the state.
\end{invariant}

\begin{proof}[Bounded Model Checking with History]
We verify using Z3 with explicit encoding:
\begin{itemize}
    \item History variable $H$ as set of bitvectors (digests)
    \item Update: $H' = H \cup \{d : \text{approved}(d)\}$
    \item Bound $B=100$ transitions, max 5 concurrent digests
    \item Property: $\neg\exists$ trace where $\text{executing}(d) \land \text{unsafe}(d) \land d \notin H$
    \item Result: \texttt{unsat} (no counterexample within bound)
\end{itemize}
\end{proof}

\begin{assumption}[Bound Sufficiency]
The bound $B=100$ exceeds any realistic episode length between system resets. For longer episodes, k-induction would be required.
\end{assumption}

\section{Cryptographic Assumptions and Threat Model}

\subsection{Explicit Cryptographic Assumptions}

\begin{assumption}[HMAC Security]
HMAC-SHA256 provides existential unforgeability under chosen message attacks (EUF-CMA). Tokens cannot be forged without knowledge of $K'$.
\end{assumption}

\begin{assumption}[Key Secrecy]
The signing key $K'$ is kept secret from the LLM and external tools. Compromise of $K'$ breaks all security guarantees.
\end{assumption}

\begin{assumption}[Digest Collision Resistance]
$\mathrm{dig}(a)$ uniquely identifies action instance $a$ with negligible collision probability.
\end{assumption}

\subsection{Threat Model: Handled vs. Not Handled}

\begin{threat}[Action-Token Mismatch - HANDLED]
Using token for $a_1$ to execute $a_2$.\\
\textbf{Mitigation:} Token binds to $\mathrm{dig}(a)$, verified at execution.\\
\textbf{Verification:} Enforced in formal model ($t.\mathrm{dig}(a) \in P$).
\end{threat}

\begin{threat}[Token Replay - PARTIALLY HANDLED]
Reusing expired or consumed token.\\
\textbf{Mitigation:} Single-use tokens with logical time expiry.\\
\textbf{Limitation:} Requires synchronized logical time; network delays not modeled.
\end{threat}

\begin{threat}[Token Exfiltration - NOT HANDLED]
Attacker reads token from logs/memory.\\
\textbf{Mitigation:} Tokens never printed or included in LLM-visible text.\\
\textbf{Assumption:} Secure storage and transmission channel.
\end{threat}

\begin{threat}[Prompt Injection - NOT HANDLED]
Adversarial prompting bypasses risk classification.\\
\textbf{Scope:} Out of scope for actuation gating verification.\\
\textbf{Mitigation (Separate):} Input sanitization, prompt hardening.
\end{threat}

\begin{threat}[TOCTOU - NOT HANDLED]
Environment changes between approval and execution.\\
\textbf{Scope:} Local TOCTOU not modeled; requires resource locking.\\
\textbf{Mitigation (Future):} Bind to resource identifiers, not paths.
\end{threat}

\section{Abstraction Soundness and Runtime Monitoring}

\subsection{Simulation Relation}

\begin{definition}[Implementation State]
$z = (\text{queues}, \text{tokens}, \text{current}, \text{logs}, \text{counters})$ with:
\begin{itemize}
    \item $\text{queues}$: maps action digests to full instances
    \item $\text{tokens}$: set of valid tokens with metadata
    \item $\text{current}$: currently executing action instance
    \item $\text{counters}$: logical time and sequence numbers
\end{itemize}
\end{definition}

\begin{definition}[Abstraction Function $\alpha$]
\[
\alpha(z) = (\text{mode}(z), \{\mathrm{dig}(a): a \in z.\text{queues}\}, 
             \mathrm{dig}(z.\text{current}), 
             \{\mathrm{dig}(a): \exists t \in z.\text{tokens}.\text{valid}(t)\}, 
             z.\text{counters}.k)
\]
\end{definition}

\begin{spec}[Simulation Preservation with Digest Check]
For each implementation transition $z_1 \to z_2$:
\begin{algorithmic}[1]
\STATE $s_1 \gets \alpha(z_1)$
\STATE $s_2 \gets \alpha(z_2)$
\STATE \textbf{assert} $(s_1, s_2) \in R_B$ \COMMENT{Abstract transition valid}
\STATE \textbf{assert} $\text{digests\_consistent}(z_1, z_2, s_1, s_2)$
\IF{violation}
    \STATE \textbf{log} full state dump
    \STATE \textbf{enter} safe mode (block all actions)
\ENDIF
\end{algorithmic}
\end{spec}

\begin{remark}[Monitor Trust Boundary]
The monitor is part of the Trusted Computing Base (TCB). We minimize its code (342 lines) and version-pin it. Property-based tests attempt to falsify monitor correctness.
\end{remark}

\section{Empirical Validation Protocols}

\subsection{Model Conformance Testing}

\begin{protocol}[Model Conformance: Gating]
\textbf{Purpose:} Verify implementation matches formal model.\\
\textbf{Method:}
\begin{enumerate}
    \item Generate 10,000 action instances from real session log distribution
    \item Risk classification: 70\% from actual classifier, 30\% random $\sim U[0,1]$
    \item Execute through full pipeline, record all transitions
    \item Check: $\alpha(z_1) \to \alpha(z_2) \in R_B$ for each transition
    \item Fuzz: inject malformed digests, expired tokens, concurrent races
\end{enumerate}
\textbf{Metrics:} Proportion of transitions preserving simulation, Wilson 95\% CI with continuity correction.
\end{protocol}

\begin{protocol}[Cryptographic Binding Correctness]
\textbf{Purpose:} Verify token-action binding.\\
\textbf{Method:}
\begin{enumerate}
    \item Generate action $a_1$, get token $t_1$
    \item Attempt to execute different action $a_2$ with $t_1$
    \item Verify rejection (must fail)
    \item Repeat 1,000 times with varied $a_1, a_2$
\end{enumerate}
\textbf{Metric:} 0/1 failures (must be 0).
\end{protocol}

\subsection{Performance Measurements}

\begin{protocol}[Memory Similarity Under Real Updates]
\textbf{Purpose:} Measure semantic drift under realistic workloads.\\
\textbf{Method:}
\begin{enumerate}
    \item Replay 100 real sessions with actual embedding model
    \item For each write: $m' = \text{update}(m, \text{new\_info}, \text{retrieval\_error})$
    \item Measure $\text{sim}(m, m') = \frac{m \cdot m'}{\|m\|\|m'\|}$
    \item Track distribution across writes
\end{enumerate}
\textbf{Metrics:} Proportion $\geq 0.95$, per-session mean similarity, bootstrap 95\% CI.
\end{protocol}

\begin{protocol}[Control Loop Stability]
\textbf{Purpose:} Assess regulator performance under disturbance.\\
\textbf{Method:}
\begin{enumerate}
    \item Define stable epoch: $\max_{t \in [i,i+9]} |c_t - \bar{c}| < 0.1$ where $\bar{c}$ is 10-step mean
    \item Run 100 episodes of 100 steps each
    \item Inject disturbances: $\sim N(0, 0.1)$ every 10 steps
    \item Count stable epochs
\end{enumerate}
\textbf{Metrics:} Proportion of stable epochs, Wilson 95\% CI.
\end{protocol}

\subsection{Statistical Methods}

\begin{itemize}
    \item \textbf{Proportions:} Wilson score interval with continuity correction
    \item \textbf{Means:} Block bootstrap (block size=10) to handle within-run correlation
    \item \textbf{Independence:} Across-run independence via fresh initialization
    \item \textbf{Software:} Python 3.11, scipy 1.11, statsmodels 0.14
\end{itemize}

\section{System Limits and Liveness Considerations}

\subsection{Explicit System Limits}

\begin{spec}[Concurrency Bound]
Maximum 5 concurrent action instances (pending + executing). Enforced by admission control.
\end{spec}

\begin{spec}[Episode Bound]
Formal verification valid for episodes $\leq 100$ transitions. Longer episodes require reset or inductive proof.
\end{spec}

\begin{spec}[Timeout Recovery]
If system enters $\text{blocked}$ due to timeout:
\begin{enumerate}
    \item Persist forensic logs
    \item Invalidate all outstanding tokens
    \item Require explicit human re-enable via secure channel
\end{enumerate}
\end{spec}

\subsection{Liveness Requirements (Not Formally Verified)}

\begin{spec}[Conditional Liveness]
If action $a$ is safe ($\neg\text{unsafe}(a)$), then execution occurs within 10 steps, assuming no system faults.
\end{spec}

\begin{spec}[Approval Responsiveness]
If action $a$ is unsafe and human operator is available, then either approval or explicit denial occurs within reasonable time (operator-dependent).
\end{spec}

\section{Appendix A: SMT Encoding of History Invariant}

\begin{lstlisting}[language=lisp,basicstyle=\scriptsize]
; State variables
(declare-fun mode (Int) Mode)
(declare-fun pending (Int) (Set Digest))
(declare-fun current (Int) Digest)
(declare-fun history (Int) (Set Digest))
(declare-fun time (Int) Int)

; Transition relation
(define-fun transition ((i Int)) Bool
  (and
    ; Mode transitions per spec
    (or
      (and (= (mode i) idle) (= (mode (+ i 1)) pending)
           (exists ((d Digest)) (and (unsafe d) (in d (pending (+ i 1))))))
      ; ... other transitions
    )
    ; History update: H' = H ∪ {approved digests}
    (forall ((d Digest))
      (=> (approved d i) (in d (history (+ i 1)))))
    ; History monotonic: H ⊆ H'
    (forall ((d Digest))
      (=> (in d (history i)) (in d (history (+ i 1)))))
  ))

; Safety property to negate
(define-fun unsafe_execution ((i Int)) Bool
  (exists ((d Digest))
    (and (executing d i) (unsafe d i) (not (in d (history i))))))

; BMC bound 100
(assert (forall ((i Int)) (=> (and (>= i 0) (< i 100)) (transition i))))
(assert (exists ((i Int)) (and (>= i 1) (<= i 100) (unsafe_execution i))))
(check-sat) ; Returns unsat
\end{lstlisting}
