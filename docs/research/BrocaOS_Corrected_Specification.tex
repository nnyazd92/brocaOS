\begin{abstract}
This document presents the corrected specification of BrocaOS, a cognitive operating system for agentic AI with formally verified action-instance safety. We provide: (1) a consistent formal model with witness-carrying execution and ever-authorized history, (2) explicit finite abstraction domains for mechanized verification, (3) k-induction proof with minimal k=1 using auxiliary invariants, (4) complete cryptographic design with clear scope boundaries, (5) runtime monitors with coverage-defined testing, and (6) reproducible empirical validation. All claims are precisely scoped with clear separation between mechanized verification (over finite abstraction), design specifications, and empirical measurements.
\end{abstract}

\section{Consistent Formal Model with Ever-Authorized History}

\subsection{Finite Abstraction Domains}

\begin{definition}[Bounded Domains for Verification Model]
For the finite abstraction $M_B$ used in mechanized verification:
\begin{itemize}
    \item $\text{id} \in \{0,\dots,2^{16}-1\}$: 16-bit identifier (per-episode)
    \item $k, \text{expiry} \in \{0,\dots,127\}$: 7-bit logical time (sufficient for bounded episodes)
    \item $\mathrm{Digests}_B = \{d_0,\dots,d_{31}\}$: 32 symbolic digests (uninterpreted)
    \item $b(a)$ abstracted as $\mathrm{digestToBytes}(d)$: mapping to symbolic byte representation
    \item Nonce abstracted as part of digest (uninterpreted freshness)
\end{itemize}
\end{definition}

\subsection{Core Type Definitions (Runtime)}

\begin{definition}[Action Instance (Runtime)]
An action instance $a = (\text{id}, \text{type}, \text{args}, \text{context\_fields}, \text{nonce})$ where runtime domains are larger but bounded by practical limits.
\end{definition}

\begin{definition}[Canonical Action Bytes]
\[
b(a) = \mathrm{JCS}(a) \quad \text{per RFC 8785 with:}
\]
\begin{itemize}
    \item No duplicate object member names
    \item Lexicographic key ordering by UTF-8 bytes
    \item Minimal escaping, no whitespace outside strings
    \item Numbers: integers as shortest representation, floats with full precision (no NaN/Infinity)
    \item UTF-8 encoding, no BOM
\end{itemize}
\end{definition}

\begin{definition}[Action Hash Digest]
\[
h(a) = \mathrm{SHA256}(b(a))
\]
\end{definition}

\begin{definition}[Classification at Generation]
\[
\mathrm{cls}(a) = 
\begin{cases}
\texttt{cls\_unsafe} & \text{if } \text{irreversible}(a) \land \text{risk}(a) > \tau \\
\texttt{cls\_safe} & \text{otherwise}
\end{cases}
\]
where $\tau=0.7$ and classification is computed in TCB.
\end{definition}

\begin{definition}[Authorization Token]
A token $t = (\mathrm{bytes}, \mathrm{expiry}, \mathrm{sig})$ where:
\begin{itemize}
    \item $\mathrm{bytes} = b(a)$: canonical action bytes
    \item $\mathrm{expiry} \in \mathbb{N}$: logical time validity bound
    \item $\mathrm{sig} = \mathrm{HMAC}_{K_{\text{sig}}}(\texttt{"token"}\|b(a)\|\text{expiry})$
\end{itemize}
Derived fields: $t.\mathrm{digest} = h(a)$, $t.\mathrm{valid}(k) \equiv \text{verifyHMAC}(t) \land k \leq \mathrm{expiry}$.
\end{definition}

\subsection{State Components with Clear Semantics}

\begin{definition}[Classification Table]
$C: \mathrm{Digests} \to \{\texttt{cls\_safe}, \texttt{cls\_unsafe}\}$ maps digests to classification at generation. For $M_B$: $C_B: \mathrm{Digests}_B \to \{\texttt{cls\_safe}, \texttt{cls\_unsafe}\}$.
\end{definition}

\begin{definition}[Ever-Authorized History]
$H_{\text{auth}} \subseteq \mathrm{Digests}$: monotone set of digests that have been authorized by a valid token. Updated when token is verified (approval time).
\[
H'_{\text{auth}} = H_{\text{auth}} \cup \{h(a) : \text{valid token verified for } a\}
\]
\end{definition}

\begin{definition}[Currently Valid Tokens]
$T_{\text{valid}} \subseteq \mathrm{Tokens}$: tokens currently valid (not expired, not consumed). Non-monotone.
\end{definition}

\subsection{Formal State Machine with Witness-Carrying Execution}

\begin{spec}[Consistent Gating State Machine]
State $S = (\text{mode}, P, E, W, H_{\text{auth}}, T_{\text{valid}}, C, k)$ where:
\begin{itemize}
    \item $\text{mode} \in \{\text{idle}, \text{pending}, \text{executing}, \text{blocked}\}$
    \item $P \subseteq \mathrm{Digests}$: pending unsafe digests
    \item $E \in \mathrm{Digests} \cup \{\bot\}$: currently executing digest
    \item $W \in \mathrm{Tokens} \cup \{\bot\}$: token witness for current execution
    \item $H_{\text{auth}}, T_{\text{valid}}, C, k$ as above
\end{itemize}

Transitions:
\begin{align*}
& (\text{idle}, P, \bot, \bot, H, T, C, k) \xrightarrow{\text{generate}(a)} \\
& \quad \begin{cases}
(\text{pending}, P\cup\{h(a)\}, \bot, \bot, H, T, C[h(a)\mapsto\texttt{cls\_unsafe}], k+1) & \text{if } \mathrm{cls}(a)=\texttt{cls\_unsafe} \\
(\text{executing}, P, h(a), \bot, H, T, C[h(a)\mapsto\texttt{cls\_safe}], k+1) & \text{if } \mathrm{cls}(a)=\texttt{cls\_safe}
\end{cases} \\
& (\text{pending}, P, \bot, \bot, H, T, C, k) \xrightarrow{\text{approve}(t)} (\text{idle}, P\setminus\{h(a)\}, \bot, \bot, H\cup\{h(a)\}, T\cup\{t\}, C, k+1) \\
& \quad \text{if } h(a) \in P \land t.\mathrm{valid}(k) \\
& (\text{idle}, P, \bot, \bot, H, T, C, k) \xrightarrow{\text{execute}(d,t)} (\text{executing}, P, d, t, H, T\setminus\{t\}, C, k+1) \\
& \quad \text{if } d \in H \land t \in T \land t.\mathrm{digest}=d \land t.\mathrm{valid}(k) \\
& (\text{executing}, P, d, t, H, T, C, k) \xrightarrow{\text{done}} (\text{idle}, P, \bot, \bot, H, T, C, k+1) \\
& \text{any state} \xrightarrow{\text{timeout}} (\text{blocked}, \emptyset, \bot, \bot, H, \emptyset, C, k+1) \quad \text{if } k > K_{\max}
\end{align*}
\end{spec}

\begin{spec}[Classification Table Consistency]
If $h(a) \in \mathrm{dom}(C)$, then $C[h(a)]$ must equal $\mathrm{cls}(a)$. No retroactive reclassification.
\end{spec}

\section{k-Induction Verification with Minimal k=1}

\subsection{Finite Abstraction Model}

\begin{definition}[Bounded Verification Model $M_B$]
$M_B = (S_B, S_0, R_B, C_B)$ where:
\begin{itemize}
    \item $S_B$: states with explicit bounds: max 5 concurrent digests, $|\mathrm{Digests}_B| = 32$, $k \in \{0,\dots,127\}$
    \item $S_0 = \{(\text{idle}, \emptyset, \bot, \bot, \emptyset, \emptyset, C_0, 0)\}$
    \item $R_B \subseteq S_B \times S_B$: transitions from Spec. 2.4 respecting bounds
    \item $C_B: \mathrm{Digests}_B \to \{\texttt{cls\_safe}, \texttt{cls\_unsafe}\}$ with consistency condition
\end{itemize}
\end{definition}

\subsection{Core Safety Invariant with Witness History}

\begin{invariant}[Action-Instance Safety with Authorization History]
For all traces $\pi = s_0 s_1 \dots$ of $M_B$:
\[
\Box\left((E = d) \land (C_B[d] = \texttt{cls\_unsafe}) \Rightarrow d \in H_{\text{auth}}\right)
\]
where $E$ is executing digest and $H_{\text{auth}}$ is ever-authorized history.
\end{invariant}

\begin{lemma}[Auxiliary Invariants for 1-Induction]
The following invariants hold in $M_B$ and are used to prove the safety invariant 1-inductive:
\begin{enumerate}
    \item $H_{\text{auth}}$ monotonicity: $H_{\text{auth}}$ only grows
    \item Classification consistency: $C_B$ unchanged after generation
    \item Token consumption: $\text{mode}=\text{executing} \land W \neq \bot \Rightarrow W.\mathrm{digest} = E$
    \item Pending classification: $d \in P \Rightarrow C_B[d] = \texttt{cls\_unsafe}$
    \item Valid token implies authorization: $t \in T_{\text{valid}} \land t.\mathrm{valid}(k) \Rightarrow t.\mathrm{digest} \in H_{\text{auth}}$
\end{enumerate}
\end{lemma}

\begin{theorem}[1-Induction Verification]
The safety invariant (3.2) holds for all traces of $M_B$ (unbounded length given finite state space).
\end{theorem}

\begin{proof}[1-Induction in Z3 with Auxiliary Invariants]
We prove using 1-induction with auxiliary invariants $\Phi = \bigwedge_{i=1}^5 \text{AuxInvariant}_i$:
\begin{enumerate}
    \item \textbf{Base case:} Show $\Phi \land \text{SafetyInvariant}$ holds in $S_0$
    \item \textbf{Inductive step:} Assume $\Phi \land \text{SafetyInvariant}$ holds in state $s_i$, prove it holds in $s_{i+1}$ for all valid transitions
\end{enumerate}
Z3 proves both steps \texttt{unsat} (no counterexample), establishing the invariant.
\end{proof}

\begin{assumption}[Finite Abstraction Sufficiency]
The bounds in $M_B$ (max 5 concurrent, 32 digests, 128 time steps) exceed any realistic episode. Runtime monitors enforce these bounds.
\end{assumption}

\section{Cryptographic Design with Clear Scope}

\subsection{Hash-Then-MAC with Scope Boundaries}

\begin{spec}[Cryptographic Design and Scope]
\begin{enumerate}
    \item \textbf{Canonicalization}: RFC 8785 removes representation ambiguity (duplicate keys, ordering, whitespace, number formatting)
    \item \textbf{Hash}: SHA-256 provides collision resistance (cryptographic assumption)
    \item \textbf{Token}: HMAC-SHA256 provides unforgeability (EUF-CMA assumption)
    \item \textbf{Scope}: Canonicalization prevents \emph{semantic-equivalence encoding collisions}, not cryptographic hash collisions
\end{enumerate}
\end{spec}

\section{Runtime Monitors with Coverage Testing}

\subsection{Monitor Specification with Coverage}

\begin{spec}[Monitor with Transition Coverage]
The monitor checks six coverage classes:
\begin{enumerate}
    \item \textbf{Token verification}: HMAC and expiry checks
    \item \textbf{History monotonicity}: $H_{\text{auth}}$ only grows
    \item \textbf{Classification consistency}: $C$ immutable after generation
    \item \textbf{Authorization check}: $d \in H_{\text{auth}}$ for unsafe execution
    \item \textbf{Token consumption}: Execution removes token from $T_{\text{valid}}$
    \item \textbf{Bound enforcement}: Max 5 concurrent, digest uniqueness
\end{enumerate}
\end{spec}

\begin{protocol}[Mutation Testing with Coverage Classes]
\textbf{Purpose:} Test monitor detection per coverage class.\\
\textbf{Method:}
\begin{enumerate}
    \item Generate 10,000 valid traces from $M_B$
    \item For each coverage class, inject corresponding violation:
    \begin{itemize}
        \item Class 1: Invalid HMAC or expired token
        \item Class 2: Remove digest from $H_{\text{auth}}$
        \item Class 3: Modify $C[d]$ after generation
        \item Class 4: Execute unsafe without $d \in H_{\text{auth}}$
        \item Class 5: Skip token consumption
        \item Class 6: Exceed concurrency bounds
    \end{itemize}
    \item Run monitor, record detection per class
\end{enumerate}
\textbf{Metrics:} Detection rate per class, Clopper--Pearson 95\% CI.
\end{protocol}

\section{Empirical Validation with Exact Statistics}

\subsection{Type and Classification Consistency}

\begin{protocol}[Complete Type Consistency Testing]
\textbf{Purpose:} Verify all type definitions and determinism.\\
\textbf{Method:}
\begin{enumerate}
    \item Generate 100,000 random action instances (i.i.d.)
    \item For each: compute $b(a)$, $h(a)$, $\mathrm{cls}(a)$
    \item Verify determinism: same $a$ yields same $b(a)$, $h(a)$, $\mathrm{cls}(a)$
    \item Verify injectivity: $a_1 \neq a_2 \Rightarrow b(a_1) \neq b(a_2)$ in sample
    \item Verify classification stability
\end{enumerate}
\textbf{Metrics:} 0 inconsistencies, exact Clopper--Pearson upper bound.
\end{protocol}

\subsection{Cryptographic Binding Correctness}

\begin{protocol}[Complete Cryptographic Binding Test]
\textbf{Purpose:} Verify token binding specificity.\\
\textbf{Method:}
\begin{enumerate}
    \item Generate $a_1$, compute $t_1 = \mathrm{token}(a_1)$
    \item For 10,000 different $a_2 \neq a_1$:
    \item Attempt $\mathrm{verify}(t_1, a_2)$
    \item Must always reject
\end{enumerate}
\textbf{Metric:} 0 false accepts, exact one-sided 95\% upper bound: $p_{\max} = 1 - 0.05^{1/10000} \approx 0.0003$.
\end{protocol}

\subsection{Statistical Methods}

\begin{itemize}
    \item \textbf{Zero-failure bounds}: One-sided Clopper--Pearson: $p_{\max} = 1 - \alpha^{1/n}$
    \item \textbf{General proportions}: Two-sided Clopper--Pearson exact
    \item \textbf{Correlated measures}: Block bootstrap with justified block size (autocorrelation analysis)
    \item \textbf{Independence}: Fresh initialization for across-run, within-run correlation handled by block bootstrap
\end{itemize}

\section{Explicit Limitations and Threat Boundaries}

\subsection{Verification Boundaries}

\begin{limitation}[Finite Abstraction Only]
k-induction proof applies to $M_B$ with explicit bounds. Runtime operation may exceed bounds; monitors enforce them.
\end{limitation}

\begin{limitation}[Classification Trust]
Safety depends on correct $\mathrm{cls}(a)$ computation. Classification is in TCB but may use LLM-derived features.
\end{limitation}

\begin{limitation}[Canonicalization Scope]
RFC 8785 prevents representation ambiguity but does not prevent adversarial construction of semantically different actions with same safety classification.
\end{limitation}

\subsection{Security Boundaries}

\begin{threat}[Prompt Injection on Classification]
Adversarial prompting may influence features used by risk classifier.\\
\textbf{Status:} Out of scope for actuation gating verification; handled by input sanitization.
\end{threat}

\begin{threat}[TOCTOU on Resource State]
Environment changes between approval and execution.\\
\textbf{Status:} Requires resource locking; out of scope for current specification.
\end{threat}

\begin{threat}[Side Channels]
Timing, memory access patterns may leak tokens or keys.\\
\textbf{Status:} Not modeled; assume secure execution environment.
\end{threat}

def canonical_action(a: Action) -> bytes:
    """RFC 8785 JSON Canonicalization Scheme."""
    # Build structured object
    obj = {
        "id": a.id,
        "type": a.type,
        "args": canonical_args(a.args),  # Recursive JCS
        "context": {k: v for k, v in sorted(a.context_fields.items())},
        "nonce": a.nonce.hex()

        reject_nan_inf=True,       # no NaN/Infinity

\section{Conclusion}

This corrected specification addresses all identified inconsistencies:
\begin{enumerate}
    \item \textbf{Consistent invariant}: Ever-authorized history $H_{\text{auth}}$ avoids token consumption mismatch
    \item \textbf{Explicit finite domains}: Bounded identifiers, time, digest count for mechanized verification
    \item \textbf{Minimal k=1 induction}: With properly chosen auxiliary invariants
    \item \textbf{Clear cryptographic scope}: RFC 8785 for representation, SHA-256/HMAC for crypto
    \item \textbf{Coverage-based testing}: Monitor testing with mutation operator classes
    \item \textbf{Exact statistics}: Clopper--Pearson with proper zero-failure bounds
\end{enumerate}

BrocaOS demonstrates that rigorous formal verification is achievable for cognitive architectures, providing instance-level authorization guarantees within explicitly bounded assumptions.
